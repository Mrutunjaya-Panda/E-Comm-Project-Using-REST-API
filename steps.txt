//c. Find the average rating of a product using aggregation pipeline in MongoDB, and then update the averageRating field of the product document with the calculated average rating. This way, we can ensure that the average rating is always up-to-date whenever a new rating is added for a product.
db.products.aggregate([
    //stage1: unwind the ratings array to deconstruct it into individual documents for each rating.
    {
        $unwind: "$ratings"
    },
    //stage2: group by name and calculate the average rating for each product using $avg operator.
    {
        $group: {
            _id: "$name", //grouping by product name
            averageRating: { $avg: "$ratings.rating" } //calculating average rating
        }
    },
])

//d. let's find out the count of ratings for each product using aggregation pipeline in MongoDB, and then update the ratingsCount field of the product document with the calculated count of ratings. This way, we can ensure that the ratings count is always up-to-date whenever a new rating is added for a product.
db.products.aggregate([
    //stage1: unwind the ratings array to deconstruct it into individual documents for each rating.
    {
        $unwind: "$ratings"
    },
    //stage2: group by name and calculate the count of ratings for each product using $sum operator.
    {
        $group: {
            _id: "$name", //grouping by product name
            ratingsCount: { $sum: 1 } //calculating count of ratings
        }
    },
])

//we can do d. using project and cond logic, let's see it.
db.products.aggregate([
    //Stage1: project name of the product and countOfRating
    {
        $project: {
            name:1,
            countOfRatings:
            {
                $cond: {
                    if: {$isArray: "$ratings"}, //check if ratings is an array
                    then: {$size: "$ratings"}, //if it is an array, get the size of the ratings array
                    else: 0 //if it is not an array, set countOfRatings to 0
                }
            }
        }
    },
    //Stage2: sort the collection based on countOfRatings in descending order to get the products with highest number of ratings at the top.
    {
        $sort: { countOfRatings: -1 }
    },
    //Stage 3: project the one with highest countOfratings, by just projecting the first document from the sorted output using $limit operator/expressions to get the product with highest number of ratings.
    {
        $limit: 1
    }
])
